function	script	MD_EMPTY_ASSERT	{
	.@ok = getarg(0);
	.@msg$ = getarg(1);

	if (.@ok)
		debugmes "[OK] " + .@msg$;
	else
		debugmes "[FAIL] " + .@msg$;

	return;
}
